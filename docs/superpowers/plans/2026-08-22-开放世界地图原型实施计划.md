# 开放世界地图原型实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一张可在电脑浏览器中探索、可识别并点击地点的二维半开放世界地图。

**Architecture:** 使用正交相机渲染代码生成的低多边形世界；输入、移动计算、世界构建、角色表现与游戏循环分离。地点与障碍来自同一份静态配置，后续独立场景可以直接消费地点标识。

**Tech Stack:** Three.js 0.179、Vite 7、原生 JavaScript 模块、CSS、node:test。

**Spec:** `docs/superpowers/specs/2026-08-22-开放世界地图原型设计.md`

## 全局约束

- 只面向电脑浏览器；键盘负责移动，鼠标负责地点交互。
- 第一版不实现独立地点内部场景、手机操作、后端、存档、音频或程序化无限地图。
- 所有视觉资源由 Three.js 几何体与 CSS 生成，不增加外部美术资源依赖。
- 正交相机、低多边形几何体、柔和低饱和配色与深棕描边共同形成二维半手绘观感。
- 每个模块只承担一个职责，纯计算逻辑必须可由 node:test 独立验证。

- [ ] **步骤四：实现最小移动逻辑**

```js
export function normalizeDirection(direction) {
  const length = Math.hypot(direction.x, direction.z);
  if (length <= 1) return { x: direction.x, z: direction.z };
  return { x: direction.x / length, z: direction.z / length };
}

function collides(position, radius, obstacles) {
  return obstacles.some((obstacle) => (
    Math.hypot(position.x - obstacle.x, position.z - obstacle.z)
      < radius + obstacle.radius
  ));
}

export function moveActor(options) {
  const direction = normalizeDirection(options.direction);
  const distance = options.speed * options.deltaSeconds;
  const candidate = {
    x: Math.max(
      options.bounds.minX + options.radius,
      Math.min(options.bounds.maxX - options.radius, options.position.x + direction.x * distance),
    ),
    z: Math.max(
      options.bounds.minZ + options.radius,
      Math.min(options.bounds.maxZ - options.radius, options.position.z + direction.z * distance),
    ),
  };
  return collides(candidate, options.radius, options.obstacles)
    ? { ...options.position }
    : candidate;
}

export function distanceToLocation(position, location) {
  return Math.hypot(position.x - location.x, position.z - location.z);
}

export function findNearbyLocation(position, locations) {
  return locations
    .map((location) => ({ location, distance: distanceToLocation(position, location) }))
    .filter((entry) => entry.distance <= entry.location.interactionRadius)
    .sort((left, right) => left.distance - right.distance)[0]?.location ?? null;
}
```

// src/world/primitives.js 的公开接口
export function createOutlinedMesh(geometry, material, options = {}) {
  const group = new THREE.Group();
  const mesh = new THREE.Mesh(geometry, material);
  group.add(mesh);
  if (options.outline !== false) {
    group.add(new THREE.LineSegments(
      new THREE.EdgesGeometry(geometry, options.thresholdAngle ?? 24),
      new THREE.LineBasicMaterial({ color: options.outlineColor ?? 0x6d5145 }),
    ));
  }
  return group;
}

export function createTree({ scale = 1, color = 0x769d65 } = {}) {
  const tree = new THREE.Group();
  const trunk = createOutlinedMesh(
    new THREE.CylinderGeometry(0.12, 0.18, 0.9, 6),
    new THREE.MeshStandardMaterial({ color: 0x8f674d, flatShading: true }),
  );
  trunk.position.y = 0.45;
  const crown = createOutlinedMesh(
    new THREE.DodecahedronGeometry(0.62, 0),
    new THREE.MeshStandardMaterial({ color, flatShading: true }),
  );
  crown.position.y = 1.15;
  tree.add(trunk, crown);
  tree.scale.setScalar(scale);
  return tree;
}

export function createMountain({ scale = 1, color = 0xb8816f } = {}) {
  const mountain = createOutlinedMesh(
    new THREE.ConeGeometry(1.5, 3.8, 7),
    new THREE.MeshStandardMaterial({ color, flatShading: true }),
  );
  mountain.scale.setScalar(scale);
  return mountain;
}

export function createCloud({ scale = 1 } = {}) {
  const cloud = new THREE.Group();
  [[-0.7, 0, 0.5], [0, 0.2, 0.8], [0.7, 0, 0.55]].forEach(([x, y, size]) => {
    const puff = new THREE.Mesh(
      new THREE.SphereGeometry(size, 10, 7),
      new THREE.MeshBasicMaterial({ color: 0xfffbef, transparent: true, opacity: 0.88 }),
    );
    puff.position.set(x, y, 0);
    cloud.add(puff);
  });
  cloud.scale.setScalar(scale);
  return cloud;
}

export function createFloatingIsland({ width, depth, height, color } = {}) {
  const island = new THREE.Group();
  const top = createOutlinedMesh(
    new THREE.CylinderGeometry(width * 0.5, width * 0.45, height * 0.35, 10),
    new THREE.MeshStandardMaterial({ color, flatShading: true }),
  );
  const underside = createOutlinedMesh(
    new THREE.ConeGeometry(width * 0.45, height, 10),
    new THREE.MeshStandardMaterial({ color: 0x8b6657, flatShading: true }),
  );
  underside.position.y = -height * 0.65;
  island.add(top, underside);
  island.scale.z = depth / width;
  return island;
}

export function createPlayer() {
  const group = new THREE.Group();
  return {
    group,
    update(deltaSeconds, direction) {
      const moving = Math.hypot(direction.x, direction.z) > 0.01;
      if (moving) group.rotation.y = Math.atan2(direction.x, direction.z);
      group.userData.walkTime = (group.userData.walkTime ?? 0) + deltaSeconds;
    },
  };
}

- [ ] **步骤三：实现键盘输入**

使用 `Set` 记录当前按键，映射方向键和 `W/A/S/D`，方向键按下时调用 `preventDefault()`，窗口失去焦点时清空状态。`dispose()` 必须移除全部事件监听器。

```js
const DIRECTIONS = {
  KeyW: [0, -1], ArrowUp: [0, -1],
  KeyS: [0, 1], ArrowDown: [0, 1],
  KeyA: [-1, 0], ArrowLeft: [-1, 0],
  KeyD: [1, 0], ArrowRight: [1, 0],
};

export function createInput(target = window) {
  const pressed = new Set();
  const onKeyDown = (event) => {
    if (!DIRECTIONS[event.code]) return;
    event.preventDefault?.();
    pressed.add(event.code);
  };
  const onKeyUp = (event) => pressed.delete(event.code);
  const onBlur = () => pressed.clear();
  target.addEventListener("keydown", onKeyDown);
  target.addEventListener("keyup", onKeyUp);
  target.addEventListener("blur", onBlur);
  return {
    getDirection() {
      let x = 0;
      let z = 0;
      for (const code of pressed) {
        x += DIRECTIONS[code][0];
        z += DIRECTIONS[code][1];
      }
      return { x: Math.sign(x), z: Math.sign(z) };
    },
    dispose() {
      target.removeEventListener("keydown", onKeyDown);
      target.removeEventListener("keyup", onKeyUp);
      target.removeEventListener("blur", onBlur);
    },
  };
}
```

- [ ] **步骤四：实现游戏运行器**

`createGame` 创建 WebGL 渲染器、正交相机、场景、射线检测器、世界与玩家。每帧时间增量限制为 `0.033` 秒，调用 `moveActor` 更新玩家位置，调用角色更新函数处理步态，通过插值更新相机和观察点，更新云层、地点光环与界面提示，最后渲染。鼠标悬停地点时高亮；点击后根据 `findNearbyLocation` 显示“可以进入”或“请先靠近”。尺寸变化时更新相机范围与渲染尺寸。

```js
export function createGame({ canvas, ui, windowTarget = window }) {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
  const scene = new THREE.Scene();
  const camera = new THREE.OrthographicCamera(-12, 12, 8, -8, 0.1, 120);
  const input = createInput(windowTarget);
  const player = createPlayer();
  const world = createWorld(scene);
  const clock = new THREE.Clock();

  function tick() {
    const deltaSeconds = Math.min(clock.getDelta(), 0.033);
    const direction = input.getDirection();
    const next = moveActor({
      position: player.group.position,
      direction,
      speed: 5,
      deltaSeconds,
      radius: 0.45,
      bounds: WORLD_BOUNDS,
      obstacles: OBSTACLES,
    });
    player.group.position.set(next.x, 0.08, next.z);
    player.update(deltaSeconds, direction);
    renderer.render(scene, camera);
  }

  return {
    start() { renderer.setAnimationLoop(tick); },
    dispose() { renderer.setAnimationLoop(null); input.dispose(); renderer.dispose(); },
  };
}
```
